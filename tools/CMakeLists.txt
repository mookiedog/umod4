cmake_minimum_required(VERSION 3.13)

# This CMakeLists.txt builds the tools required by other parts of this project.
# All tools are compiled to run on the host's OS.
#
# This project is meant to be included as a CMake ExternalProject. The CMakeLists.txt
# that adds this ExternalProject must define the following variables:
#   LIB_DIR (the path to the directory containing the shared source files)
#   TOOLS_INSTALL_DIR (the path to where the tools should be installed after getting built by this file)
#   PYTHON_GEN_DIR (the path to the directory where the generated Python modules get written)

if (NOT DEFINED LIB_DIR)
    message(FATAL ": Variable LIB_DIR must be defined!")
endif()

if (NOT DEFINED TOOLS_INSTALL_DIR)
    message(FATAL ": Variable TOOLS_INSTALL_DIR must be defined!")
endif()

if (NOT DEFINED PYTHON_GEN_DIR)
    message(FATAL_ERROR ": Variable PYTHON_GEN_DIR must be defined!")
endif()

project(tools)

# This works because there is only 1 tool that needs to be built
# Get the checksum tool built:
add_executable(checksum
    src/checksum.cpp
)

target_include_directories(checksum PRIVATE ${LIB_DIR}/inc)

add_custom_command(
    OUTPUT ${TOOLS_INSTALL_DIR}/checksum
    DEPENDS checksum
    VERBATIM
    COMMAND ${CMAKE_COMMAND} -E copy checksum ${TOOLS_INSTALL_DIR}/checksum
)

add_custom_target(checksum-install ALL
    DEPENDS ${TOOLS_INSTALL_DIR}/checksum
)

####################################################################################################
# Generate the Python modules that are derived from log_ids.h:
#   Logsyms.py: a constant for every log ID (generated by h2py.py)
#   LogMeta.py: a name and units for the log IDs that have them (generated by gen_logmeta.py)
#
# Both are plain modules written directly into PYTHON_GEN_DIR. Python code that needs them
# (decodelog.py, the EP build) is pointed at that directory: nothing gets installed into the
# virtual environment.

set(SOURCE_FILES
    ${CMAKE_CURRENT_SOURCE_DIR}/../lib/inc/log_ids.h
)

# The C header to Python constant converter script:
set(PYTHON_SCRIPT ${CMAKE_CURRENT_SOURCE_DIR}/src/h2py.py)

set(LOGSYMS_PY_FILE ${PYTHON_GEN_DIR}/Logsyms.py)

add_custom_command(
    OUTPUT ${LOGSYMS_PY_FILE}
    DEPENDS ${SOURCE_FILES} ${PYTHON_SCRIPT} ${CMAKE_CURRENT_LIST_FILE}
    COMMENT "Generating Logsyms.py"
    COMMAND ${CMAKE_COMMAND} -E make_directory ${PYTHON_GEN_DIR}
    COMMAND ${Python3_EXECUTABLE} ${PYTHON_SCRIPT} --class-name Logsyms -o ${LOGSYMS_PY_FILE} ${SOURCE_FILES}
    VERBATIM
)

# Generate LogMeta.py (name/units metadata) from the _NAME/_UNITS macros in log_ids.h.
# Runs after Logsyms.py so the dependency chain is: log_ids.h -> Logsyms.py -> LogMeta.py.
set(LOGMETA_SCRIPT ${CMAKE_CURRENT_SOURCE_DIR}/src/gen_logmeta.py)
set(LOGMETA_PY_FILE ${PYTHON_GEN_DIR}/LogMeta.py)

add_custom_command(
    OUTPUT ${LOGMETA_PY_FILE}
    DEPENDS ${SOURCE_FILES} ${LOGMETA_SCRIPT} ${LOGSYMS_PY_FILE}
    COMMENT "Generating LogMeta.py"
    COMMAND ${Python3_EXECUTABLE} ${LOGMETA_SCRIPT} ${SOURCE_FILES} -o ${LOGMETA_PY_FILE}
    VERBATIM
)

# Create a custom target that depends on both generated modules
add_custom_target(update_log_symbols ALL
    DEPENDS ${LOGSYMS_PY_FILE} ${LOGMETA_PY_FILE}
)
